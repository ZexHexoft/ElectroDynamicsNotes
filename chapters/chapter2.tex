\section{真空中的静电场}

\begin{definition}[静电场]
    \begin{equation}
        \mathbf{E}=\frac{1}{4\pi \varepsilon _0}\int_V{\frac{\rho \left( \mathbf{r}' \right)}{\rcurs^2}\hrcurs\mathrm{d}\tau '}
    \end{equation}
\end{definition}

\begin{theorem}[高斯定理]
    \begin{gather}
        \oint_S{\mathbf{E}\cdot \mathrm{d}\mathbf{a}}=\frac{Q}{\varepsilon _0}\\
        \nabla \cdot \mathbf{E}=\frac{\rho}{\varepsilon _0}
    \end{gather}
\end{theorem}

\begin{definition}[静电势]
    \begin{equation}
        V(\vec r):=-\int_O^r{\mathbf{E}\cdot \mathrm{d}\mathbf{l}}
    \end{equation}
\end{definition}

\begin{corollary}
    \begin{equation}
        \vec E = - \nabla V
    \end{equation}
\end{corollary}

\begin{definition}[泊松方程]
    \begin{equation}
        \nabla^2 V = - \frac{\rho}{\varepsilon_0} 
    \end{equation}
\end{definition}

对无电荷区域，退化为

\begin{definition}[拉普拉斯方程]
    \begin{equation}
        \nabla^2 V = 0 \label{eq:Laplace_equation}
    \end{equation}
\end{definition}

\begin{theorem}[边界条件]\label{boundary_statement}
    \begin{gather}
        \mathbf{E}_{\text{上}}-\mathbf{E}_{\text{下}}=\frac{\sigma}{\varepsilon _0}\hat{\mathbf{n}}\\
        \frac{\partial V_{\text{上}}}{\partial n}-\frac{\partial V_{\text{下}}}{\partial n}=-\frac{\sigma}{\varepsilon _0}
    \end{gather}
\end{theorem}

我们熟知分立电荷的能量

\begin{theorem}
    \begin{equation}
        W=\frac{1}{2}\sum_{i=1}^n{q_iV\left( \mathbf{r}_i \right)}
    \end{equation}
\end{theorem}

接下来我们计算连续电荷分布的能量\footnote{注意\cref{eq:diff_rule3}的逆运用}

\begin{theorem}
    \begin{align}
        W&=\frac{1}{2}\int{\rho V\mathrm{d}\tau}
        \\
        &=\frac{\varepsilon _0}{2}\int{\left( \nabla \cdot \mathbf{E} \right) V\mathrm{d}\tau}\nonumber
        \\
        &=\frac{\varepsilon _0}{2}\int{\left[ \nabla \cdot \left( V\mathbf{E} \right) -\mathbf{E}\cdot \nabla V \right] \mathrm{d}\tau}\nonumber
        \\
        &=\frac{\varepsilon _0}{2}\left( \int_V{E^2\mathrm{d}\tau}+\oint_S{V\mathbf{E}\cdot \mathrm{d}\mathbf{a}} \right) \nonumber
        \\
        &=\frac{\varepsilon _0}{2}\int_V{E^2\mathrm{d}\tau}
    \end{align}
\end{theorem}

\section{介质中的静电场}

考虑线性电介质\footnote{希望在这里不加介绍地引入电极化矢量不会给读者造成困扰.}:

\begin{theorem}[束缚电荷]
    \begin{gather}
        Q_b:=\int{\mathrm{d}^3\mathbf{x}\rho _b}=-\oint{\mathbf{P}\cdot \mathrm{d}\mathbf{a}}\\
        \rho_b = -\nabla \cdot \vec P\\
        \sigma_b = \hat{\vec n}\cdot \vec P
    \end{gather}
\end{theorem}

由连续性方程

$$
\frac{\partial \rho_b}{\partial t}+\nabla \cdot \vec J_b = 0
$$

我们有如下推论: 

\begin{corollary}[束缚电流密度]
    \begin{equation}
        \vec J_b = \frac{\partial \vec P}{\partial t}
    \end{equation}
\end{corollary}

我们引入电位移矢量 $\vec{D}$:

\begin{definition}[电位移矢量]
    \begin{equation}
        \vec{D} = \varepsilon_0 \vec{E}+\vec{P} =  \varepsilon \vec{E}
    \end{equation}
\end{definition}

则有
\begin{corollary}[介质中的泊松方程]
    \begin{equation}
        \nabla \cdot \vec{D} = \rho
    \end{equation}
\end{corollary}

\section{唯一性定理与静电镜像法}

我们先介绍两种静电学中常用的边界条件:

\begin{itemize}
    \item 狄利克雷(Dirichlet)边界条件: 已知静电势在边界 $S$上的取值.
    \item 诺伊曼(Neumann)边界条件: 已知静电势在边界 $S$法向偏微商的取值.
\end{itemize}

接下来我们给出在这三种边界条件下的唯一性定理:

\begin{theorem}[静电唯一性定理] 
    考虑一个区域 $V$(可能有导体存在)中有给定的电荷密度 $\rho$, 如果每一个导体的总电荷已经给定, 在狄利克雷(Dirichlet)边界条件或诺伊曼(Neumann)边界条件下, 静电边值问题的解是唯一的.
\end{theorem}

\begin{center}
    \incfig{唯一性定理}
    \captionof{figure}{} \label{fig2}
\end{center}

\begin{proof}
    由于我们不关心导体内部的电势, 我们下面按\cref{fig2}中虚线表示的空间和表面积分.

    取区域内满足泊松方程的两个连续函数解 $\Phi_1, \Phi_2$, 我们定义 $\Psi = \Phi_1 - \Phi_2$, 其在区域内满足
    $$
    \nabla ^2\Psi = 0
    $$
    在不与导体接壤的边界上满足
    \begin{align*}
        &\Psi = 0,\quad \text{(Dirichlet)}\\
        &\frac{\partial \Psi}{\partial n} = 0,\quad \text{(Neumann)}
    \end{align*}
    对于与导体 $C$接壤的边界, 我们有
    $$
    \Psi = \Phi_1 - \Phi_2 = c = Const.
    $$
    同时由于导体所带电量恒定, 我们有
    $$
    \oint_C \frac{\partial \Psi}{\partial n}\mathrm{d}S = 0
    $$
    考虑恒等式
    $$
    \nabla \cdot \left( \Psi \nabla \Psi\right) = \left( \nabla \Psi \right) ^2 + \Psi\nabla^2 \Psi 
    $$
    因而有
    \begin{align*}
        \int_V \mathrm{d}^3 \vec{x}\left( \nabla \Psi \right)^2 &= \oint_S \Psi \left( \nabla \Psi\right)\cdot \mathrm{d}\vec{S} - \int_V\mathrm{d}^3\vec{x}\Psi\nabla^2 \Psi \\
        &= \oint_S \Psi \left(\frac{\partial \Psi}{\partial n}\right)\mathrm{d}S - \int_V\mathrm{d}^3\vec{x}\Psi\nabla^2 \Psi \\
    \end{align*}
    在两种边界条件下，右侧均为零.因此我们有
    $$
    \nabla \Psi = 0
    $$
    从而在区域内有
    $$
    \Psi = Const.
    $$
    即$\Phi_1, \Phi_2$最多相差一个常数, 在经典条件下对产生的电场没有影响, 即此时静电边值问题的解是唯一的.
\end{proof}

唯一性定理给予了我们给出"猜测解"的可能性, 对这一定理最为熟知的应用应该是静电镜像法.

经典镜像法的基本原理是: 在所求区域之外引入适当的镜像电荷分布，使原问题在边界上的条件（如导体表面等势、导体总电荷给定等）得到满足，同时保持区域内真实电荷分布不变. 由唯一性定理，区域内的电场唯一，因此该构造给出的电场就是原问题的解.

需要注意，镜像电荷必须放在所求区域之外，否则会改变区域内的源分布: 若导体是绝缘且带给定总电荷，还需检查或附加镜像以满足总电荷条件.

在此我们不展示常见的例子(无限大导体平面、接地导体球等), 我们给出一个来自Jackson电动力学的有趣例子

    \centering

    \caption{导体球在均匀电场中的极化}\label{fig:method_of_image}

\begin{example}
    考虑如\cref{fig:method_of_image}的构型, 设导体电势为 $0$, 我们使用电像法证明此时空间中的电势分布为
    \begin{equation}
        \Phi = -E_0(r-\frac{a^3}{r^2})\cos \theta  \quad (r>a)
    \end{equation}
\end{example}

\begin{figure}[h]
    \centering
    \includegraphics[width=0.8\textwidth]{./figures/method_of_image_a.pdf}
    \caption{由位于无穷远处的异种电荷生成的匀强电场}\label{fig:method_of_image_a}
\end{figure}

\begin{figure}[h]
    \centering
    \includegraphics[width=0.8\textwidth]{./figures/method_of_image_b.pdf}
    \caption{应用镜像法后}\label{fig:method_of_image_b}
\end{figure}


\begin{proof}
    参考\cref{fig:method_of_image_a}, 对于 $R\gg 1$我们可以估计
    \begin{equation}
        E_0 \simeq \frac{2Q}{4\pi\varepsilon_0 R^2} \label{eq:image_1}
    \end{equation}
    在保证 $Q/R^2$不变的情况下令$R\to \infty$, 则上式变为等式, 即该构型可以生成匀强电场.

    从而我们应用接地导体球关于电像法的经典结论(见\cref{fig:method_of_image_b})给出
    \begin{equation}
        \begin{split}
            \Phi &= \frac{Q/4\pi\varepsilon_0}{(r^2+R^2+2rR\cos\theta)^{1/2}}
            -\frac{Q/4\pi\varepsilon_0}{(r^2+R^2-2rR\cos\theta)^{1/2}}\\
            &\quad -\frac{aQ/4\pi\varepsilon_0}{R\left(r^2+\displaystyle \frac{a^4}{R^2}+\displaystyle \frac{2a^2r}{R^2}\cos\theta\right)^{1/2}}
            + \frac{aQ/4\pi\varepsilon_0}{R\left(r^2+\displaystyle \frac{a^4}{R^2}-\displaystyle \frac{2a^2r}{R^2}\cos\theta\right)^{1/2}}
        \end{split}
    \end{equation}

    依照这个构型, 我们有 $a < r \ll R$以及 $Q \sim R^2$, 从而我们写出

    \begin{equation}
        \Phi = \frac{1}{4\pi\varepsilon_0}\left( -\frac{2Q}{R^2}r\cos\theta+\frac{2Q}{R^2}\frac{a^3}{r^2}\cos\theta \right) 
    \end{equation}

    进而利用\cref{eq:image_1}给出

    \begin{equation}
        \Phi = -E_0\left( r-\frac{a^3}{r^2} \right)\cos\theta 
    \end{equation}
\end{proof}
